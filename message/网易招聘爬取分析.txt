如何用 Scrapy 获取“下一页”URL？
1. 分析请求方式
网易招聘的分页是前端异步加载，不是传统的 a 标签跳转。

2. Scrapy 爬取建议
直接请求 API，而不是模拟点击“下一页”。

示例 Scrapy 代码
import scrapy

class Hr163Spider(scrapy.Spider):
    name = "hr163"
    allowed_domains = ["hr.163.com"]
    start_urls = [
        "https://hr.163.com/api/hr163/position/query.do?curPage=1&pageSize=10"
    ]

    def parse(self, response):
        data = response.json()
        # 处理职位数据
        for job in data['result']['list']:
            yield job

        # 判断是否有下一页
        cur_page = data['result']['curPage']
        total_page = data['result']['totalPageCount']
        if cur_page < total_page:
            next_page = cur_page + 1
            next_url = f"https://hr.163.com/api/hr163/position/query.do?curPage={next_page}&pageSize=10"
            yield scrapy.Request(next_url, callback=self.parse)

import scrapy

class Hr163Spider(scrapy.Spider):
    name = "hr163"
    allowed_domains = ["hr.163.com"]
    start_urls = [
        "https://hr.163.com/api/hr163/position/query.do?curPage=1&pageSize=10"
    ]

    def parse(self, response):
        data = response.json()
        # 处理职位数据
        for job in data['result']['list']:
            yield job

        # 当前页码
        cur_page = data['result']['curPage']
        # 总页数
        total_page = data['result']['totalPageCount']

        # 判断是否还有下一页
        if cur_page < total_page:
            next_page = cur_page + 1
            next_url = f"https://hr.163.com/api/hr163/position/query.do?curPage={next_page}&pageSize=10"
            yield scrapy.Request(next_url, callback=self.parse)
